# gfd-wsrelay — WebSocket<->UDP bridge for the GFD Battlegrounds WASM client (replaces the box's gfd-wsudprelay + nginx /gfd-ws).
# Single listener in --path-mode-port: wss://ws.rg.okemily.com/gfd-ws/<udp-port>; TLS is the Certificate Manager wildcard
# (*.rg.okemily.com) on the edge Ingress. Relays to the stable game servers' in-cluster UDP Service. BackendConfig lifts the
# LB's 30s default timeout so a WebSocket survives a whole match.
pod name=gfd-wsrelay namespace=emily image=us-central1-docker.pkg.dev/project-d24a71e9-2daf-4b2d-917/emily/gfd-wsudprelay:0.1.0 secret=redgarden-env var=/app/var run=/run/redgarden replicas=1

container name=relay command=/wsudprelay args=--path-mode-port,8081,--udp-host,redgarden-stable-udp.emily.svc.cluster.local,--matchmaker-port,8778,--game-port-start,8300,--game-port-count,20 port=8081 cpu=100 mem=128
backendconfig name=ws-long timeout=3600
service ports=8081 backend-config=ws-long
